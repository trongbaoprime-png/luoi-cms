function doGet(e) {
  e = e || { parameter: {} };
  const month = Number(e.parameter.month || new Date().getMonth() + 1);
  const callback = e.parameter.callback || '';
  const fresh = String(e.parameter.fresh || '') === '1';

  let json = fresh ? null : readCacheJson(month);

  if (!json) {
    const data = getDashboardData(month);
    json = JSON.stringify(data);
    saveCacheJson(month, json);
  }

  if (callback) {
    return ContentService.createTextOutput(`${callback}(${json})`)
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  return ContentService.createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}


function refreshMonthCache(month) {
  month = Number(month || new Date().getMonth() + 1);
  const data = getDashboardData(month);
  saveCacheJson(month, JSON.stringify(data));
}

function refreshAllDashboardCache() {
  for (let month = 1; month <= 12; month++) {
    const data = getDashboardData(month);
    saveCacheJson(month, JSON.stringify(data));
  }
}

function refreshCurrentMonthCache() {
  const month = new Date().getMonth() + 1;
  const data = getDashboardData(month);
  saveCacheJson(month, JSON.stringify(data));
}

function refreshCurrentAndNearbyCache() {
  const current = new Date().getMonth() + 1;
  const months = [current - 1, current, current + 1].filter(m => m >= 1 && m <= 12);

  months.forEach(month => {
    const data = getDashboardData(month);
    saveCacheJson(month, JSON.stringify(data));
  });
}

function setupFastCacheTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'refreshCurrentAndNearbyCache') {
      ScriptApp.deleteTrigger(t);
    }
  });

  ScriptApp.newTrigger('refreshCurrentAndNearbyCache')
    .timeBased()
    .everyMinutes(5)
    .create();
}

function setupAutoRefreshCache() {
  setupFastCacheTrigger();
}

function clearDashboardCache() {
  const props = PropertiesService.getScriptProperties();
  const keys = props.getKeys();

  keys.forEach(key => {
    if (key.indexOf('cache_file_month_') === 0) {
      const fileId = props.getProperty(key);
      try {
        DriveApp.getFileById(fileId).setTrashed(true);
      } catch (err) {}
    }

    if (key.indexOf('cache_file_month_') === 0 || key.indexOf('cache_time_month_') === 0) {
      props.deleteProperty(key);
    }
  });
}

function readCacheJson(month) {
  const props = PropertiesService.getScriptProperties();
  const fileId = props.getProperty('cache_file_month_' + month);
  const time = Number(props.getProperty('cache_time_month_' + month) || 0);

  if (!fileId) return null;
  if (Date.now() - time > CONFIG.cacheMinutes * 60 * 1000) return null;

  try {
    return DriveApp.getFileById(fileId).getBlob().getDataAsString();
  } catch (err) {
    return null;
  }
}

function saveCacheJson(month, json) {
  const props = PropertiesService.getScriptProperties();
  const key = 'cache_file_month_' + month;
  const oldFileId = props.getProperty(key);
  const fileName = 'dashboard_cache_' + getSheetName(month) + '.json';

  try {
    if (oldFileId) {
      DriveApp.getFileById(oldFileId).setContent(json);
    } else {
      const file = DriveApp.createFile(fileName, json, MimeType.PLAIN_TEXT);
      props.setProperty(key, file.getId());
    }

    props.setProperty('cache_time_month_' + month, String(Date.now()));
  } catch (err) {
    const file = DriveApp.createFile(fileName, json, MimeType.PLAIN_TEXT);
    props.setProperty(key, file.getId());
    props.setProperty('cache_time_month_' + month, String(Date.now()));
  }
}

function getDashboardData(month) {
  const sheetName = getSheetName(month);
  const records = [];
  const logs = [];

  CONFIG.telesales.forEach(t => {
    try {
      const ss = SpreadsheetApp.openByUrl(t.url);
      const sheet = ss.getSheetByName(sheetName);

      if (!sheet) {
        logs.push({ name: normalizeTelesale(t.name), status: '❌', msg: 'Không có sheet ' + sheetName });
        return;
      }

      const lastRow = sheet.getLastRow();

      if (lastRow < 3) {
        logs.push({ name: normalizeTelesale(t.name), status: '⚠️', msg: 'Chưa có dữ liệu' });
        return;
      }

      const rows = sheet.getRange(3, 2, lastRow - 2, 17).getDisplayValues();

      rows.forEach(row => {
        const item = parseRow(row, t.name, month);
        if (item) records.push(item);
      });

      const s = summarizeServer(records.filter(r => r.telesale === normalizeTelesale(t.name)));

      logs.push({
        name: normalizeTelesale(t.name),
        status: '✅',
        msg:
          `${s.customers} dòng hợp lệ`
          + ` | Check-in: ${s.checkin}`
          + ` | Đậu: ${s.pass}`
          + ` | Rớt: ${s.fail}`
          + ` | Khách mới: ${s.newCustomer}`
          + ` | Khách cũ: ${s.oldCustomer}`
          + ` | Khách cũ PS: ${s.oldCustomerPS}`
          + ` | DT Khách cũ PS: ${formatNumber(s.oldCustomerPSRevenue)}`
          + ` | DT: ${formatNumber(s.revenue)}`
      });

    } catch (err) {
      logs.push({ name: normalizeTelesale(t.name), status: '❌', msg: String(err) });
    }
  });

  // Đọc thêm dữ liệu ĐẶT HẸN từ file tổng DATHEN, vùng A3:J
  try {
    const dathen = getDathenRecords(month);
    records.push.apply(records, dathen.records);
    logs.push(dathen.log);
  } catch (err) {
    logs.push({ name: 'DATHEN', status: '❌', msg: String(err) });
  }

  return {
    success: true,
    cachedAt: new Date().toLocaleString('vi-VN'),
    sheetName,
    rawCount: records.length,
    logs,
    records
  };
}



function getDathenRecords(month) {
  const ss = SpreadsheetApp.openByUrl(CONFIG.dathenUrl);
  const sheet = ss.getSheetByName(CONFIG.dathenSheetName);
  const records = [];

  if (!sheet) {
    return { records, log: { name: 'DATHEN', status: '❌', msg: 'Không có sheet DATHEN' } };
  }

  const lastRow = sheet.getLastRow();
  if (lastRow < 3) {
    return { records, log: { name: 'DATHEN', status: '⚠️', msg: 'Chưa có dữ liệu' } };
  }

  // Đọc đúng vùng A3:J. Dùng getValues để giữ Date thật ở cột A,
  // đồng thời getDisplayValues để các cột text/SĐT không bị mất số 0 đầu.
  const values = sheet.getRange(3, 1, lastRow - 2, 10).getValues();
  const displays = sheet.getRange(3, 1, lastRow - 2, 10).getDisplayValues();

  let totalRows = 0;
  let validDateRows = 0;

  values.forEach((row, idx) => {
    totalRows++;
    const displayRow = displays[idx] || [];
    const item = parseDathenRow(row, displayRow, month);
    if (getInputDateKey(row[0]) || getInputDateKey(displayRow[0])) validDateRows++;
    if (item) records.push(item);
  });

  return {
    records,
    log: {
      name: 'DATHEN',
      status: '✅',
      msg: `${records.length} đặt hẹn hợp lệ trong ${getSheetName(month)} | Có ngày: ${validDateRows}/${totalRows} dòng A3:J`
    }
  };
}

function parseDathenRow(row, displayRow, month) {
  // A ngày nhập liệu | B họ tên | C SĐT | D nguồn | E chi nhánh | F DV chính | G DV chi tiết | H diễn giải | I sale hẹn | J cũ/mới
  displayRow = displayRow || row;

  const inputDate = row[0] || displayRow[0];
  const name = displayRow[1] || row[1];
  const phone = displayRow[2] || row[2];

  if (!name && !phone) return null;

  const dateKey = getInputDateKey(inputDate) || getInputDateKey(displayRow[0]);
  if (!dateKey) return null;

  // Lọc đúng tháng/năm theo ngày đặt hẹn ở cột A, không dùng ngày check-in.
  const y = Number(dateKey.slice(0, 4));
  const m = Number(dateKey.slice(5, 7));
  if (y !== Number(CONFIG.year) || m !== Number(month)) return null;

  const source = normalizeSource(displayRow[3] || row[3] || 'Chưa rõ');
  const branch = normalizeBranch(displayRow[4] || row[4] || 'Chưa rõ');
  const service = normalizeService(displayRow[5] || displayRow[6] || row[5] || row[6] || 'Chưa rõ');
  const telesale = normalizeTelesale(displayRow[8] || row[8] || 'Chưa rõ');
  const oldNew = normalizeText(displayRow[9] || row[9]);

  return {
    type: 'DATHEN',
    fullName: name,
    phone: phone,
    colB: name,
    colC: phone,
    colD: displayRow[3] || row[3],
    colE: displayRow[4] || row[4],
    colH: displayRow[8] || row[8],
    appointment: 1,
    customer: 0,
    telesale,
    source,
    sourceGroup: getSourceGroup(source),
    branch,
    branchGroup: getBranchGroup(branch),
    service,
    serviceGroup: getServiceGroup(service),
    ctkm: 'ĐẶT HẸN',

    // Ngày đặt hẹn độc lập với ngày check-in.
    appointmentDate: dateKey,
    appointmentMonth: dateKey.slice(0, 7),
    checkinDate: '',
    checkinMonth: '',
    isMonthNote: false,

    checkin: 0,
    result: 'Đặt hẹn',
    pass: 0,
    fail: 0,

    revenue: 0,
    newRevenue: 0,
    oldRevenue: 0,
    oldCustomerPSRevenue: 0,
    actualRevenue: 0,
    actualNewRevenue: 0,
    actualOldRevenue: 0,
    caTheo: 0,
    caTheoRevenue: 0,
    vietKieuRevenue: 0,
    koMkt: 0,

    // Không cộng vào Khách mới/Khách cũ của check-in để tránh làm sai bảng hiệu suất.
    appointmentNew: oldNew.includes('MOI') ? 1 : 0,
    appointmentOld: oldNew.includes('CU') ? 1 : 0,
    newCustomer: 0,
    oldCustomer: 0,
    oldCustomerPS: 0
  };
}

function getInputDateKey(value) {
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    const y = value.getFullYear();
    const m = String(value.getMonth() + 1).padStart(2, '0');
    const d = String(value.getDate()).padStart(2, '0');
    return `${y}-${m}-${d}`;
  }

  return getCheckinDateKey(value);
}

function isHeaderOrGuideRow(row) {
  const cName = normalizeText(row[1]);   // Cột C - Họ tên
  const dPhone = normalizeText(row[2]);  // Cột D - SĐT
  const eSource = normalizeText(row[3]); // Cột E - Nguồn
  const fBranch = normalizeText(row[4]); // Cột F - Chi nhánh
  const gService = normalizeText(row[5]); // Cột G - DVU
  const hDate = normalizeText(row[6]);   // Cột H - Ngày checkin

  if (!cName && !dPhone && !eSource && !fBranch && !gService && !hDate) return true;

  // Bỏ dòng tiêu đề của sheet cá nhân: HỌ TÊN | SỐ ĐT | NGUỒN | CHI NHÁNH | DVU | NGÀY CHECK IN
  if (cName.includes('HO TEN') || dPhone.includes('SO DT') || eSource === 'NGUON' || fBranch.includes('CHI NHANH') || gService === 'DVU') return true;
  if (hDate.includes('NGAY') && hDate.includes('CHECK')) return true;

  // Bỏ các dòng note/hướng dẫn nếu lỡ nằm trong vùng đọc
  if (cName.includes('MUON NOTE') || dPhone.includes('MUON NOTE') || hDate.includes('NGAY KHACH DEN')) return true;

  return false;
}

function parseRow(row, telesale, month) {
  if (isHeaderOrGuideRow(row)) return null;

  const name = row[1];
  const phone = row[2];

  if (!name && !phone) return null;

  const source = normalizeSource(row[3] || 'Chưa rõ');
  const branch = normalizeBranch(row[4] || 'Chưa rõ');
  const service = normalizeService(row[5] || 'Chưa rõ');

  const hValue = row[6];   // Cột H
  const jValue = row[8];   // Cột J
  const kValue = row[9];   // Cột K
  const lValue = row[10];  // Cột L
  const mValue = row[11];  // Cột M
  const nValue = row[12];  // Cột N
  const oValue = row[13];  // Cột O
  const pValue = row[14] || 'Không có CTKM'; // Cột P

  const revenue = parseRevenueJ(jValue);
  const hasSpecificDate = isSpecificDateH(hValue);
  const hasMonthNote = isMonthNoteH(hValue);
  const isOld = isOldCustomerByColumnO(oValue);

  const checkin = hasSpecificDate ? 1 : 0;
  const result = getResult(jValue, checkin);
  const caTheo = isValidCaTheo(lValue) ? 1 : 0;
  const actualMoney = parseMoney(mValue);
  const isNewCustomer = hasSpecificDate && !isOld;
  const isOldCustomer = hasSpecificDate && isOld;

  // Quy tắc nghiệp vụ TDS: Sheet nhân viên tháng m thì toàn bộ doanh thu/checkin thuộc 100% về tháng m của sheet đó
  let rawDate = getCheckinDateKey(hValue);
  const targetYM = `${CONFIG.year}-${String(month || new Date().getMonth() + 1).padStart(2, '0')}`;
  if (rawDate && rawDate.length === 10) {
    rawDate = `${targetYM}-${rawDate.slice(8, 10)}`;
  } else {
    rawDate = `${targetYM}-01`;
  }

  return {
    type: 'CHECKIN',
    fullName: name,
    phone: phone,
    colC: name,
    colD: phone,
    appointment: 0,
    customer: 1,
    telesale: normalizeTelesale(telesale),
    source,
    sourceGroup: getSourceGroup(source),
    branch,
    branchGroup: getBranchGroup(branch),
    service,
    serviceGroup: getServiceGroup(service),
    ctkm: pValue,

    checkinDate: rawDate,
    checkinMonth: targetYM,
    isMonthNote: hasMonthNote,

    checkin,
    result,
    pass: result === 'Đậu' ? 1 : 0,
    fail: result === 'Rớt' ? 1 : 0,

    revenue,
    newRevenue: isNewCustomer ? revenue : 0,
    oldRevenue: isOldCustomer ? revenue : 0,
    oldCustomerPSRevenue: hasMonthNote ? revenue : 0,

    actualRevenue: isNewCustomer || isOldCustomer ? actualMoney : 0,
    actualNewRevenue: isNewCustomer ? actualMoney : 0,
    actualOldRevenue: isOldCustomer ? actualMoney : 0,

    caTheo,
    caTheoRevenue: caTheo ? parseMoney(lValue) : 0,

    nnCustomer: isNguoiNuocNgoai(nValue) ? 1 : 0,
    nnRevenue: isNguoiNuocNgoai(nValue) ? revenue : 0,
    vietKieuRevenue: isVietKieu(nValue) ? revenue : 0,
    koMkt: isKoMkt(kValue) ? 1 : 0,

    newCustomer: isNewCustomer ? 1 : 0,
    oldCustomer: isOldCustomer ? 1 : 0,
    oldCustomerPS: hasMonthNote ? 1 : 0
  };
}

function summarizeServer(records) {

    t.oldCustomerPS += Number(r.oldCustomerPS || 0);
    t.revenue += Number(r.revenue || 0);

    t.oldRevenue += Number(r.oldRevenue || 0);

function isOldCustomerByColumnO(value) {
  const text = normalizeText(value);
  if (!text) return false;

  return (
    text === 'OLD' ||
    text === 'CU' ||
    text.includes('OLD') ||
    text.includes('KHACH CU') ||
    text.includes('KH CU')
  );
}

function getCheckinDateKey(value) {
  const raw = String(value || '').trim();
  const text = normalizeText(raw);

  if (!raw) return '';
  if (text.includes('TRONG THANG')) return '';

  const m = raw.match(/(\d{1,2})[\/\-.](\d{1,2})(?:[\/\-.](\d{2,4}))?/);
  if (!m) return '';

  let d = Number(m[1]);
  let mo = Number(m[2]);
  let y = m[3] ? String(m[3]) : String(CONFIG.year);

  if (y.length === 2) y = '20' + y;

  if (mo > 12 && d <= 12) {
    const temp = d;
    d = mo;
    mo = temp;
  }

  if (mo < 1 || mo > 12) return '';
  if (d < 1 || d > 31) return '';

  return `${y}-${String(mo).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
}

function getCheckinMonthKey(value) {
  const raw = String(value || '').trim();
  const text = normalizeText(raw);

  if (!raw) return '';

  const note = text.match(/TRONG THANG\s*(\d{1,2})/);

  if (note) {
    const mo = Number(note[1]);
    if (mo < 1 || mo > 12) return '';
    return `${CONFIG.year}-${String(mo).padStart(2, '0')}`;
  }

  const m = raw.match(/(\d{1,2})[\/\-.](\d{1,2})(?:[\/\-.](\d{2,4}))?/);
  if (!m) return '';

  let d = Number(m[1]);
  let mo = Number(m[2]);
  let y = m[3] ? String(m[3]) : String(CONFIG.year);

  if (y.length === 2) y = '20' + y;

  if (mo > 12 && d <= 12) {
    const temp = d;
    d = mo;
    mo = temp;
  }

  if (mo < 1 || mo > 12) return '';

  return `${y}-${String(mo).padStart(2, '0')}`;
}

function isMonthNoteH(value) {
  const text = normalizeText(value);
  return /TRONG THANG\s*\d{1,2}/.test(text);
}

function isSpecificDateH(value) {
  const text = normalizeText(value);
  if (!text) return false;
  if (isMonthNoteH(value)) return false;
  return /\d{1,2}[\/\-.]\d{1,2}/.test(text);
}

function normalizeBranch(value) {
  const text = normalizeText(value);

  if (text.includes('DA LAT')) return 'ĐÀ LẠT';
  if (text.includes('TAY NINH')) return 'TÂY NINH';
  if (text.includes('DA NANG')) return 'ĐÀ NẴNG';
  if (text.includes('QUY NHON')) return 'QUY NHƠN';
  if (text.includes('BINH PHUOC')) return 'BÌNH PHƯỚC';

  if (text.includes('THU DUC')) return 'THỦ ĐỨC';
  if (text.includes('BINH CHANH')) return 'BÌNH CHÁNH';
  if (text.includes('BINH THANH')) return 'BÌNH THẠNH';
  if (text.includes('GO VAP')) return 'GÒ VẤP';
  if (text.includes('HOC MON')) return 'HÓC MÔN';
  if (text.includes('LANDMARK')) return 'LANDMARK';
  if (text.includes('LBB')) return 'LBB';
  if (text.includes('QUAN 1')) return 'QUẬN 1';
  if (text.includes('QUAN 3')) return 'QUẬN 3';
  if (text.includes('QUAN 7')) return 'QUẬN 7';
  if (text.includes('TAN BINH')) return 'TÂN BÌNH';
  if (text.includes('TAN PHU')) return 'TÂN PHÚ';
  if (text.includes('TEN LUA')) return 'TÊN LỬA';
  if (text.includes('VINH LOC B')) return 'VĨNH LỘC B';

function isRot(value) {
  const text = normalizeText(value);
  return text === 'R' || text === 'ROT' || text === 'RỚT';
}

function isDauValue(value) {
  const text = normalizeText(value);

  if (!text) return false;
  if (isRot(text)) return false;
  if (text === 'THEO') return true;
  if (text === '0') return true;

  return text.replace(/[^\d]/g, '') !== '';
}

function getResult(jValue, checkin) {
  if (!checkin) return 'Chưa có';

  if (isRot(jValue)) return 'Rớt';
  if (isDauValue(jValue)) return 'Đậu';

  return 'Chưa có';
}

function parseRevenueJ(value) {
  const text = normalizeText(value);

  if (!text) return 0;
  if (isRot(text)) return 0;
  if (text === 'THEO') return 0;
  if (text === '0') return 0;

  const number = text.replace(/[^\d]/g, '');

  return number ? Number(number) : 0;
}

  return text ? Number(text) : 0;
}

  );

function isVietKieu(value) {

  const text = normalizeText(value);

  if (!text) return false;

  // Người nước ngoài => KHÔNG tính Việt kiều
  if (
    text === 'NN' ||
    text.includes('NGUOI NUOC NGOAI') ||
    text.includes('FOREIGN')
  ) {
    return false;
  }

  // Việt kiều
  if (text === 'VK') return true;
  if (text.includes('VK')) return true;
  if (text.includes('VIET KIEU')) return true;

  // Danh sách quốc gia
  const countries = [
    'MY',
    'USA',
    'CANADA',
    'UC',
    'ANH',
    'PHAP',
    'DUC',
    'NHAT',
    'HAN',
    'HAN QUOC',
    'DAI LOAN',
    'SINGAPORE',
    'UC CHAU',
    'NEW ZEALAND',
    'THUY SI',
    'MALAYSIA'
  ];

  return countries.some(c => text.includes(c));
}

function normalizeTelesale(value){

  const t = normalizeText(value);

  if (t === 'CHANG' || t === 'TRANG') {
    return 'TRANG';
  }

  if (t === 'LIEU') {
    return 'LIỂU';
  }

  if (t === 'TELE' || t === 'TELEVOI') {
    return 'TELEVOI';
  }

  return String(value || '').trim().toUpperCase();
}

function normalizeText(value) {
  return String(value || '')
    .toUpperCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/Đ/g, 'D')
    .replace(/\s+/g, ' ')
    .trim();
}

function formatNumber(value) {
  return Number(value || 0).toLocaleString('vi-VN');
}
